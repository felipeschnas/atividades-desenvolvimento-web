-- Exercício 10 - MySQL

-- Criar a base de dados
CREATE DATABASE escola;
USE escola;

-- Criar a tabela
CREATE TABLE alunos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    idade INT
);

-- Inserir dados
INSERT INTO alunos (nome, email, idade) VALUES ('Ana Souza', 'ana@email.com', 20);
INSERT INTO alunos (nome, email, idade) VALUES ('Bruno Lima', 'bruno@email.com', 22);
INSERT INTO alunos (nome, email, idade) VALUES ('Carla Mendes', 'carla@email.com', 19);

-- Visualizar dados
SELECT * FROM alunos;

-- Alterar dados
UPDATE alunos SET idade = 21 WHERE id = 1;
UPDATE alunos SET email = 'bruno.lima@email.com' WHERE nome = 'Bruno Lima';
SELECT * FROM alunos;

-- Apagar dados
DELETE FROM alunos WHERE id = 3;
SELECT * FROM alunos;

-- Apagar tabela
DROP TABLE alunos;

-- Apagar base de dados
DROP DATABASE escola;
